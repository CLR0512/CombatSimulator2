# Firemaking: Chk's Normal bonfire has time left but isn't lit, so it gives nothing. Lit on B it adds 1% XP gained,
# which reaches the result's XP and nothing in the fight; out again is the save's.
cs2 page firemaking a
expect reply "firemaking changeable groups=1 Your bonfire: Normal Bonfire, not burning"
expect reply "Normal 0/1 was=0 modifiers=1"
cs2 whatif check bonfire=Redwood
expect reply "bonfire=Redwood accepted was=out"
cs2 whatif DragonsPass/Drake bonfire=Normal seed=1
cs2 wait timeout=9
cs2 wait timeout=9
wait 1500
cs2 compare show metric=xp:main
expect reply "bonfire=Normal accepted was=out"
expect reply "DragonsPass/Drake  A=13,019 [12,768, 13,271]  B=13,138 [12,884, 13,392]"
cs2 compare show metric=kills
expect reply "DragonsPass/Drake  A=59.09 [57.91, 60.28]  B=59.09 [57.91, 60.28]  d=0"
cs2 side b bonfire=Normal
expect reply "bonfire=Normal accepted was=out"
wait 2500
cs2 page firemaking b
expect reply "Normal 1/1 was=0 modifiers=1"
cs2 side b bonfire=out
expect reply "bonfire=out no-change was=out"
